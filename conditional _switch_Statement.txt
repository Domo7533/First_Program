using System;
					
public class Program
{
	public static void Main()
	{
		string day = "Friday";
		switch (day) {
			case "Monday":
				Console.WriteLine("Start of week"); 
				break; 
			case "Friday": 
				Console.WriteLine("Weekend soon"); 
				break;
			default: 
				Console.WriteLine("Regular day"); 
				break; 
		}
	}
}
