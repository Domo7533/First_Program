using System;
					
public class Program
{
	public static void Main()
	{
		for (int i = 1; i <= 3; i++) // Outer loop
        {
            for (int j = 1; j <= 3; j++) // Inner loop
            {
                Console.Write($"{i * j}\t");
            }
			  Console.WriteLine(); // Move to the next line after each row
        }
	}
}
